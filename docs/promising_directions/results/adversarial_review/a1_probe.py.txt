class Forced(CausalSwitchingGame):
    """open-loop calendar: query forced at slots (G_t = time), actions fully adaptive"""
    def __init__(s,c,m,cal): super().__init__(c,m); s.cal=frozenset(cal)
    def choices(s,h):
        t=len(h)+1; rem=s.config.budget-sum(q for _,q in h)
        return tuple(product((0,1),(1,) if t in s.cal and rem else (0,)))

def cell(p):
    cfg=GameConfig(horizon=p['T'],budget=p['B'],delay=p['d'],alpha=F(p['alpha']),query_cost=F(p['cq']),switch_cost=F(p['cs']),max_nodes=400000)
    m=RegimeModel(transition01=F(p['t01']),transition10=F(p['t10']),emission0=F(p['e0']),emission1=F(p['e1']),prior1=F(p['prior1']))
    t0=time.time()
    J=F(FirstPromotionDecomposedGame(cfg,m).solve()[0]['calibrated_total_cost'])
    slots=range(1,p['T']-p['d']+1)
    best={'skip':None,'forced':None}
    for k in range(p['B']+1):
        for cal in combinations(slots,k):
            c=F(CalendarOpportunityGame(cfg,m,cal).solve()[0]['calibrated_total_cost'])
            if best['skip'] is None or c<best['skip'][0]: best['skip']=(c,cal)
            if k==p['B'] or True:
                c2=F(Forced(cfg,m,cal).solve()[0]['calibrated_total_cost'])
                if best['forced'] is None or c2<best['forced'][0]: best['forced']=(c2,cal)
    return dict(tag=p.get("tag"),p=p,J=float(J),skip=float(best['skip'][0]),skipcal=best['skip'][1],forced=float(best['forced'][0]),forcedcal=best['forced'][1],
      g_skip=float((best['skip'][0]-J)/J),g_forced=float((best['forced'][0]-J)/J),sec=time.time()-t0)

def base(**kw):
    p=dict(T=7,B=3,d=1,alpha='1/10',cq='1/100',cs='1/10',t01='1/10',t10='1/10',e0='1/10',e1='9/10',prior1='1/2'); p.update(kw); return p
if __name__=="__main__":
    which=sys.argv[1]
    if which=="time":
        print(cell(base(T=6,B=2,d=1)))
    else:
        cells=json.loads(open(which).read())
        with mp.Pool(int(sys.argv[2])) as pool:
            for r in pool.imap_unordered(cell,cells):
                print(json.dumps(r),flush=True)
